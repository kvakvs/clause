cmake_minimum_required(VERSION 3.28)
project(CompiledErlangExample LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Use the configured Clang/clang-cl driver for this ABI example")
endif()
if(NOT IS_ABSOLUTE "${GENERATED_DIR}")
    message(FATAL_ERROR "Set GENERATED_DIR to the absolute directory containing emitted native objects")
endif()
get_filename_component(repository "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# A build that already compiled the runtime may pass its exported targets file (Clause's tests do); otherwise the
# runtime is built here from the repository.
if(RUNTIME_TARGETS)
    include("${repository}/cmake/ProjectOptions.cmake")
    include("${RUNTIME_TARGETS}")
else()
    set(CLAUSE_BUILD_COMPILER OFF CACHE BOOL "" FORCE)
    set(CLAUSE_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${repository}" runtime-build)
    if(TARGET clause_llvm_sdk)
        message(FATAL_ERROR "The generated-program consumer must remain LLVM-free")
    endif()
endif()
set(suffix .o)
if(WIN32)
    set(suffix .obj)
endif()
set(objects)
foreach(module IN ITEMS 616e73776572 636c69656e74)
    set(object "${GENERATED_DIR}/clausev1_${module}__0${suffix}")
    if(NOT EXISTS "${object}")
        message(FATAL_ERROR "Emit both native modules first: missing ${object}")
    endif()
    list(APPEND objects "${object}")
endforeach()
add_executable(compiled_modules harness.cpp ${objects})
clause_project_options(compiled_modules)
target_link_libraries(compiled_modules PRIVATE Clause::generated_program)
set_target_properties(compiled_modules PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
